<script>
function calculateTotal() {
const rows = document.querySelectorAll('#table-body tr');
let grandTotal = 0;

rows.forEach(row => {
const qty = parseFloat(row.querySelector('.qty-input').value) || 0;
const price = parseFloat(row.querySelector('.price-input').value) || 0;
const total = qty * price;
row.querySelector('.row-total').innerText = 'Rs. ' + total.toFixed(2);
grandTotal += total;
});

document.getElementById('grand-total-val').innerText = 'Rs. ' + grandTotal.toFixed(2);

const received = parseFloat(document.getElementById('amt-received').value) || 0;
const balance = grandTotal - received;
const balContainer = document.getElementById('balance-container');
const balVal = document.getElementById('balance-val');

balVal.innerText = 'Rs. ' + Math.abs(balance).toFixed(2);

if (balance > 0) {
balContainer.className = "balance-box balance-due";
document.getElementById('lbl-balance').innerText = "Balance Due:";
} else {
balContainer.className = "balance-box balance-clear";
document.getElementById('lbl-balance').innerText = "Balance Clear (Advance):";
}
}

function addRow() {
const tbody = document.getElementById('table-body');
const rowCount = tbody.children.length + 1;
const tr = document.createElement('tr');
tr.innerHTML = `
<td>${rowCount}</td>
<td><input type="number" class="qty-input" oninput="calculateTotal()" min="0"></td>
<td><input type="number" class="price-input" oninput="calculateTotal()" min="0"></td>
<td class="row-total">Rs. 0.00</td>
`;
tbody.appendChild(tr);
}

function clearForm() {
document.getElementById('table-body').innerHTML = `
<tr>
<td>1</td>
<td><input type="number" class="qty-input" oninput="calculateTotal()" min="0"></td>
<td><input type="number" class="price-input" oninput="calculateTotal()" min="0"></td>
<td class="row-total">Rs. 0.00</td>
</tr>
`;
document.getElementById('work-desc').value = '';
document.getElementById('amt-received').value = '';
calculateTotal();
}

function saveHisab() {
const grandTotal = document.getElementById('grand-total-val').innerText;
const work = document.getElementById('work-desc').value || 'Unnamed Work';
const balance = document.getElementById('balance-val').innerText;

if (grandTotal === 'Rs. 0.00') {
alert('Please enter some calculations first!');
return;
}

const histList = document.getElementById('history-list');
const div = document.createElement('div');
div.className = 'history-item';
div.innerHTML = `
<span class="delete-hist" onclick="this.parentElement.remove()">X</span>
<strong>${work}</strong><br>
Total: ${grandTotal} | Balance: ${balance}
`;
histList.prepend(div);
clearForm();
}

function openCalculator() { document.getElementById('calcModal').style.display = 'flex'; }
function closeCalculator() { document.getElementById('calcModal').style.display = 'none'; }
function pressCalc(val) { document.getElementById('calc-screen').value += val; }
function clearCalc() { document.getElementById('calc-screen').value = ''; }
function solveCalc() {
try {
document.getElementById('calc-screen').value = eval(document.getElementById('calc-screen').value);
} catch(e) {
document.getElementById('calc-screen').value = 'Error';
}
}
